function toggleAdditionType() { const additionType = document.getElementById('additionType').value; const creamFatInput = document.getElementById('creamFat'); const creamFatLabel = document.getElementById('creamFatLabel'); const addLabel = document.getElementById('addLabel'); if (additionType === 'fat') { creamFatInput.value = '100'; creamFatInput.disabled = true; creamFatLabel.innerText = 'Vetgehalte vloeibaar vet (%):'; addLabel.innerText = 'Hoeveelheid vloeibaar vet toevoegen:'; } else { creamFatInput.value = ''; creamFatInput.placeholder = 'bijv. 43,00'; creamFatLabel.innerText = 'Vetgehalte toe te voegen room (%):'; addLabel.innerText = 'Hoeveelheid room toevoegen:'; // Schakel alleen in als het gewenste vetgehalte HOGER is dan het gemeten vetgehalte const currentFat = parseFloat(document.getElementById('currentFat').value) || 0; const targetFat = parseFloat(document.getElementById('targetFat').value) || 0; creamFatInput.disabled = (targetFat <= currentFat && targetFat > 0); } calculate(); } function calculate() { const batchSize = parseFloat(document.getElementById('batchSize').value) || 0; const currentFat = parseFloat(document.getElementById('currentFat').value) || 0; const targetFat = parseFloat(document.getElementById('targetFat').value) || 0; const additionTypeSelect = document.getElementById('additionType'); const creamFatInput = document.getElementById('creamFat'); const additionType = additionTypeSelect.value; // --- INACTIEF MAKEN VAN DE VELDEN BIJ VERDUNNEN --- if (currentFat > targetFat && targetFat > 0) { additionTypeSelect.disabled = true; creamFatInput.disabled = true; } else { additionTypeSelect.disabled = false; // Alleen instellen op disabled als er gekozen is voor 'vloeibaar vet' if (additionType === 'fat') { creamFatInput.disabled = true; } else { creamFatInput.disabled = false; } } const creamFat = parseFloat(creamFatInput.value) || 0; const rowAddIngredient = document.getElementById('rowAddIngredient'); const rowAddWater = document.getElementById('rowAddWater'); let addAmount = 0; let waterToAdd = 0; let stepsHtml = ''; // Vetgehalte verhogen (room/vet toevoegen) if (currentFat > 0 && targetFat > currentFat) { rowAddIngredient.style.display = 'flex'; rowAddWater.style.display = 'none'; if (creamFat > targetFat) { addAmount = ((targetFat - currentFat) / (creamFat - targetFat)) * batchSize; const currentFatKg = (batchSize * currentFat) / 100; const addFatKg = (addAmount * creamFat) / 100; const totalFatKgVal = currentFatKg + addFatKg; const totalBatchVal = batchSize + addAmount; const ingredientName = additionType === 'fat' ? 'vloeibaar vet' : 'room'; stepsHtml = `
Stap 1: Huidige hoeveelheid vet
${batchSize} kg × ${currentFat.toFixed(2)}% = ${currentFatKg.toFixed(2)} kg vet
Stap 2: Benodigde ${ingredientName} berekenen
(${targetFat.toFixed(2)}% - ${currentFat.toFixed(2)}%) / (${creamFat.toFixed(2)}% - ${targetFat.toFixed(2)}%) × ${batchSize} kg = ${Math.round(addAmount)} kg ${ingredientName}
Stap 3: Vet uit toevoeging
${Math.round(addAmount)} kg × ${creamFat.toFixed(2)}% = ${addFatKg.toFixed(2)} kg vet
Stap 4: Totale batch & vet
Totale batch na correctie: ${batchSize} kg + ${Math.round(addAmount)} kg = ${Math.round(totalBatchVal)} kg
Totaal vet: ${currentFatKg.toFixed(2)} kg + ${addFatKg.toFixed(2)} kg = ${Math.round(totalFatKgVal)} kg vet
`; } else { stepsHtml = '

Het vetgehalte van de toevoeging moet hoger zijn dan het gewenste vetgehalte.

'; } } // Vetgehalte verlagen (verdunnen met water) else if (currentFat > targetFat && targetFat > 0) { rowAddIngredient.style.display = 'none'; rowAddWater.style.display = 'flex'; waterToAdd = ((currentFat / targetFat) - 1) * batchSize; const currentFatKg = (batchSize * currentFat) / 100; const totalBatchVal = batchSize + waterToAdd; stepsHtml = `
Stap 1: Huidige hoeveelheid vet
${batchSize} kg × ${currentFat.toFixed(2)}% = ${currentFatKg.toFixed(2)} kg vet
Stap 2: Water toevoegen voor verdunning
( ${currentFat.toFixed(2)}% / ${targetFat.toFixed(2)}% - 1 ) × ${batchSize} kg = ${Math.round(waterToAdd)} kg water
Stap 3: Totale batch & vet
Totale batch na correctie: ${batchSize} kg + ${Math.round(waterToAdd)} kg = ${Math.round(totalBatchVal)} kg
Totaal vet blijft ongewijzigd: ${Math.round(currentFatKg)} kg vet
`; } else { rowAddIngredient.style.display = 'flex'; rowAddWater.style.display = 'flex'; stepsHtml = '

Vul de gegevens in om de berekening in stappen te zien.

'; } const totalBatch = batchSize + addAmount + waterToAdd; // Weergave bijwerken document.getElementById('addValue').innerText = Math.round(addAmount) + ' kg'; document.getElementById('waterToAdd').innerText = Math.round(waterToAdd) + ' kg'; document.getElementById('totalBatch').innerText = Math.round(totalBatch) + ' kg'; document.getElementById('calcSteps').innerHTML = stepsHtml; }
